2. Add GET route for Factiva config page:
   ```python
   @router.get("/factiva", response_class=HTMLResponse)
   def get_factiva_config():
       """Serve Factiva query configuration page."""
       db = SessionLocal()
       try:
           config = db.query(FactivaConfig).filter(FactivaConfig.id == 1).first()
           if not config:
               # Seed default if missing (shouldn't happen — startup migration handles this)
               config = FactivaConfig(
                   id=1, industry_codes="i82,i832", company_codes="MM",
                   keywords="insurance reinsurance", page_size=25, enabled=True
               )
               db.add(config)
               db.commit()
               db.refresh(config)

           template = jinja_env.get_template("admin/factiva.html")
           return HTMLResponse(template.render(
               config=config,
               success=None,
               error=None
           ))
       finally:
           db.close()
   ```

3. Add POST route for saving Factiva config:
   ```python
   @router.post("/factiva", response_class=HTMLResponse)
   def update_factiva_config(
       request: Request,
       industry_codes: str = Form(""),
       company_codes: str = Form(""),
       keywords: str = Form(""),
       page_size: int = Form(25),
       enabled: bool = Form(False),
   ):
       """Update Factiva query configuration."""
       db = SessionLocal()
       try:
           config = db.query(FactivaConfig).filter(FactivaConfig.id == 1).first()
           if not config:
               config = FactivaConfig(id=1)
               db.add(config)

           # Validate page_size
           if page_size not in (10, 25, 50, 100):
               page_size = 25

           # Clean inputs — strip whitespace from comma-separated codes
           config.industry_codes = ",".join(c.strip() for c in industry_codes.split(",") if c.strip())
           config.company_codes = ",".join(c.strip() for c in company_codes.split(",") if c.strip())
           config.keywords = keywords.strip()
           config.page_size = page_size
           config.enabled = enabled
           config.updated_at = datetime.utcnow()

           db.commit()
           db.refresh(config)

           logger.info("factiva_config_updated",
                        industry_codes=config.industry_codes,
                        company_codes=config.company_codes,
                        keywords=config.keywords,
                        page_size=config.page_size,
                        enabled=config.enabled)

           template = jinja_env.get_template("admin/factiva.html")
           return HTMLResponse(template.render(
               config=config,
               success="Factiva configuration saved successfully.",
               error=None
           ))
       except Exception as e:
           logger.error("factiva_config_update_failed", error=str(e))
           template = jinja_env.get_template("admin/factiva.html")
           return HTMLResponse(template.render(
               config=config if config else FactivaConfig(id=1),
               success=None,
               error=f"Failed to save configuration: {str(e)}"
           ))
       finally:
           db.close()
   ```

Note: The `enabled` checkbox in HTML forms only sends a value when checked. Handle this by checking for the presence of the field. Use `Form(False)` as default — FastAPI will set it to False when the checkbox is unchecked (no value sent). Alternatively, add a hidden field approach if Form(False) doesn't work with checkboxes — test this.

The hidden field trick (`<input type="hidden" name="enabled" value="false">`) ensures the form always sends an "enabled" value.
